// Lógica de cálculo bidireccional (Precio <-> Porcentaje) window.calcularMecanica = (input, tipo) => { const fila = input.closest('tr'); const costo = parseFloat(fila.querySelector('.p-costo').value) || 0; const margen = fila.querySelector('.p-margen'); const precio = fila.querySelector('.p-precio'); if (costo <= 0) return; // Necesitamos el costo para cualquier cálculo if (tipo === 'porcentaje') { // Calcular Precio basado en Porcentaje const mVal = parseFloat(margen.value) || 0; const resultadoPrecio = costo + (costo * (mVal / 100)); precio.value = resultadoPrecio.toFixed(2); } else if (tipo === 'precio') { // Calcular Porcentaje basado en Precio const pVal = parseFloat(precio.value) || 0; if (pVal > 0) { const resultadoMargen = ((pVal - costo) / costo) * 100; margen.value = resultadoMargen.toFixed(2); } } };
